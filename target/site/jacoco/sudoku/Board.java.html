package sudoku;

public class Board {

    private int[][] grid = new int[9][9];

    public boolean isValidMove(int row, int col, int value) {
        if (value < 1 || value > 9) {
            return false;
        }

        for (int c = 0; c < 9; c++) {
            if (grid[row][c] == value) {
                return false;
            }
        }

        for (int r = 0; r < 9; r++) {
            if (grid[r][col] == value) {
                return false;
            }
        }

        int startRow = (row / 3) * 3;
        int startCol = (col / 3) * 3;

        for (int r = startRow; r < startRow + 3; r++) {
            for (int c = startCol; c < startCol + 3; c++) {
                if (grid[r][c] == value) {
                    return false;
                }
            }
        }

        return true;
    }

    public boolean isComplete() {
        return false;
    }

    public int[][] getGrid() {
        return grid;
    }
}
